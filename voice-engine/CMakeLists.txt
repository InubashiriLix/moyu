cmake_minimum_required(VERSION 3.24)
project(novel_voice VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
option(NV_BUILD_ENGINE "Build CUDA speech engine (disable for text-only tests)" ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(nv_audio STATIC src/audio/audio.cpp src/audio/cache.cpp)
target_include_directories(nv_audio PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src/audio)
target_link_libraries(nv_audio PUBLIC Threads::Threads OpenSSL::Crypto ${CMAKE_DL_LIBS} m)
add_library(nv_text STATIC src/text/text.cpp)
target_include_directories(nv_text PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src/text)
target_compile_options(nv_text PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  add_executable(nv_text_test src/text/text_test.cpp)
  target_link_libraries(nv_text_test PRIVATE nv_text)
  add_test(NAME text COMMAND nv_text_test)
  add_executable(nv_audio_test src/audio/audio_test.cpp)
  target_link_libraries(nv_audio_test PRIVATE nv_audio)
  add_test(NAME audio COMMAND nv_audio_test)
  add_executable(nv_fake_worker tests/fake_worker.cpp)
  set_target_properties(nv_fake_worker PROPERTIES OUTPUT_NAME nv-worker RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/test-bin)
  add_library(nv_test_session SHARED src/engine/session.cpp src/engine/session_run.cpp src/engine/worker.cpp)
  target_link_libraries(nv_test_session PRIVATE nv_text nv_audio Threads::Threads ${CMAKE_DL_LIBS})
  set_target_properties(nv_test_session PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/test-bin)
  add_executable(nv_session_test tests/session_test.cpp)
  target_link_libraries(nv_session_test PRIVATE nv_test_session)
  add_dependencies(nv_session_test nv_fake_worker)
  set_target_properties(nv_session_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/test-bin BUILD_RPATH "$ORIGIN")
  add_test(NAME session COMMAND nv_session_test)
  set_tests_properties(session PROPERTIES TIMEOUT 20)
endif()
if(NV_BUILD_ENGINE)
  # The pinned backend has Windows-only playback counters exposed unguarded
  # on Linux. Keep the small, reproducible compatibility fix in this repo.
  execute_process(COMMAND git -C ${CMAKE_CURRENT_SOURCE_DIR}/vendor/qwen3 apply --reverse --check
    ${CMAKE_CURRENT_SOURCE_DIR}/patches/linux-playback-metrics.patch
    RESULT_VARIABLE NV_PATCH_APPLIED OUTPUT_QUIET ERROR_QUIET)
  if(NOT NV_PATCH_APPLIED EQUAL 0)
    execute_process(COMMAND git -C ${CMAKE_CURRENT_SOURCE_DIR}/vendor/qwen3 apply
      ${CMAKE_CURRENT_SOURCE_DIR}/patches/linux-playback-metrics.patch RESULT_VARIABLE NV_PATCH_RESULT)
    if(NOT NV_PATCH_RESULT EQUAL 0)
      message(FATAL_ERROR "Could not apply pinned Linux backend compatibility patch")
    endif()
  endif()
  execute_process(COMMAND git -C ${CMAKE_CURRENT_SOURCE_DIR}/vendor/qwen3 apply --reverse --check
    ${CMAKE_CURRENT_SOURCE_DIR}/patches/cuda-device-compatibility.patch
    RESULT_VARIABLE NV_COMPAT_APPLIED OUTPUT_QUIET ERROR_QUIET)
  if(NOT NV_COMPAT_APPLIED EQUAL 0)
    execute_process(COMMAND git -C ${CMAKE_CURRENT_SOURCE_DIR}/vendor/qwen3 apply
      ${CMAKE_CURRENT_SOURCE_DIR}/patches/cuda-device-compatibility.patch RESULT_VARIABLE NV_COMPAT_RESULT)
    if(NOT NV_COMPAT_RESULT EQUAL 0)
      message(FATAL_ERROR "Could not apply CUDA compatibility probe patch")
    endif()
  endif()
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES "89-real;120a-real" CACHE STRING "Deployment GPU architectures")
  endif()
  set(QWEN3_TTS_CUDA ON CACHE BOOL "" FORCE)
  set(GGML_CUDA ON CACHE BOOL "" FORCE)
  set(GGML_CUDA_GRAPHS ON CACHE BOOL "CUDA graph acceleration is required for the speech worker" FORCE)
  set(QWEN3_TTS_COREML OFF CACHE BOOL "" FORCE)
  set(QWEN3_TTS_BUILD_STREAMING_CLI OFF CACHE BOOL "" FORCE)
  set(QWEN3_TTS_ENABLE_MODEL_TESTS OFF CACHE BOOL "" FORCE)
  set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  find_package(CUDAToolkit REQUIRED)
  add_subdirectory(vendor/qwen3 EXCLUDE_FROM_ALL)
  set_property(TARGET ggml-cuda APPEND PROPERTY BUILD_RPATH "${CUDAToolkit_LIBRARY_DIR}")
  add_subdirectory(src/engine)
endif()
